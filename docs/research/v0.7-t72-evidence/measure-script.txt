def run(args,log,timeout=300):
    p=subprocess.run([str(a) for a in args],env=ENV,cwd=ROOT,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,encoding="utf8",errors="replace",timeout=timeout)
    (ROOT/log).write_text(p.stdout,encoding="utf8")
    if p.returncode: raise RuntimeError(str(args)+" failed, see "+str(ROOT/log)+"\n"+p.stdout[-1800:])
    return p.stdout
def py(v): return v/("Scripts/python.exe" if sys.platform=="win32" else "bin/python")

def fixture(dest):
    stream=b"BT /F1 16 Tf 40 100 Td (T72 PDF text) Tj ET"
    objs=[b"<< /Type /Catalog /Pages 2 0 R >>",b"<< /Type /Pages /Kids [3 0 R] /Count 1 >>",b"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 300 200] /Resources << /Font << /F1 4 0 R >> >> /Contents 5 0 R >>",b"<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>",b"<< /Length "+str(len(stream)).encode()+b" >>\nstream\n"+stream+b"\nendstream"]
    data=b"%PDF-1.4\n"; offsets=[0]
    for i,o in enumerate(objs,1): offsets.append(len(data)); data+=str(i).encode()+b" 0 obj\n"+o+b"\nendobj\n"
    xref=len(data); data+=b"xref\n0 6\n0000000000 65535 f \n"+b"".join(("%010d 00000 n \n"%o).encode() for o in offsets[1:]); data+=b"trailer\n<< /Size 6 /Root 1 0 R >>\nstartxref\n"+str(xref).encode()+b"\n%%EOF\n"
    dest.write_bytes(data)
